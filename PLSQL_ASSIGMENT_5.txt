6)--IF-THEN-ELSIF-ELSE.

DECLARE 
       AGE NUMBER(10):=:AGE;
BEGIN
     IF(AGE > 18) THEN 
         DBMS_OUTPUT.PUT_LINE('TICKET PRICE 200');
     ELSIF(AGE < 10)THEN 
         DBMS_OUTPUT.PUT_LINE('TICKET IS PRICE 100:');
     ELSIF(AGE > 30) THEN
         DBMS_OUTPUT.PUT_LINE('TICKTE PRICE IS 300:');
     ELSE
         DBMS_OUTPUT.PUT_LINE('TICKET  REGULAR PRICE IS 150:');
     END IF;
END;








7)--Input three no and find out maximum and minimum from it.

DECLARE 
       A NUMBER(10):=:A;
       B NUMBER(10):=:B;
       C NUMBER(10):=:C;
BEGIN
     IF (A<B AND B>C) THEN
               DBMS_OUTPUT.PUT_LINE('MAX='||B);
      
     ELSIF(B<C AND C>A) THEN 
            DBMS_OUTPUT.PUT_LINE('MAX:='||C);
        
     ELSE
            DBMS_OUTPUT.PUT_LINE('MAX:='||A);
         
     END IF;
       
       IF (A>B AND B<C) THEN
               DBMS_OUTPUT.PUT_LINE('MIN='||B);
      
     ELSIF(B>C AND C<A) THEN 
            DBMS_OUTPUT.PUT_LINE('MIN:='||C);
        
     ELSE
            DBMS_OUTPUT.PUT_LINE('MIN:='||A);
         
     END IF;
END ;

(10)-- print first 10 numbers using basic loop.. exit loop.

      END LOOP;

14-- print first 10 prime number using while loop --
DECLARE 
    N NUMBER := 2; 
    CNT NUMBER; 
BEGIN 
    WHILE N <= 30 LOOP 
        CNT := 0; 
 
        FOR I IN 1..N LOOP 
            IF MOD(N, I) = 0 THEN 
                CNT := CNT + 1; 
            END IF; 
        END LOOP; 
 
        IF CNT = 2 THEN 
            DBMS_OUTPUT.PUT_LINE(N); 
        END IF; 
 
        N := N + 1; 
    END LOOP; 
END;
/

15-- accept a string from user and display reverse of it --

DECLARE
    STR VARCHAR2(100) := '&STR';
    REV VARCHAR2(100) := '';
BEGIN
    FOR I IN REVERSE 1..LENGTH(STR)
    LOOP
        REV := REV || SUBSTR(STR,I,1);
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Original String: ' || STR);
    DBMS_OUTPUT.PUT_LINE('Reverse String: ' || REV);
END;
/

17--  to print number pyramid --
DECLARE
BEGIN
    FOR I IN 1..5
    LOOP
        FOR J IN 1..I
        LOOP
            DBMS_OUTPUT.PUT(J || ' ');
        END LOOP;

        DBMS_OUTPUT.NEW_LINE;
    END LOOP;
END;

19/* program to insert 10 row into square table. (Looping to insert
record into specified table) */

CREATE TABLE SQUARE
(
    NUM NUMBER,
    SQR NUMBER
);
BEGIN
    FOR I IN 1..10
    LOOP
        INSERT INTO SQUARE
        VALUES(I, I*I);
    END LOOP;

    COMMIT;
END;
/
SELECT * FROM SQUARE;


20/* a program to update row of square table. ( Looping to update
specified table.) */
BEGIN
    FOR I IN 1..10
    LOOP
        UPDATE SQUARE
        SET SQR = I * I
        WHERE NUM = I;
    END LOOP;

    COMMIT;
END;
/


21/* program to delete row from square table. (Looping to delete
record from specified table) */
BEGIN
    FOR I IN 1..10
    LOOP
        DELETE FROM SQUARE
        WHERE NUM = I;
    END LOOP;

    COMMIT;
END;
/
